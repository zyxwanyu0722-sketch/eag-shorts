np.choose(i, [p, p, t, v, v, q]) return np.stack([r, g, b], 1) def sstep(a, b, x): t = np.clip((x - a) / (b - a), 0.0, 1.0) return t * t * (3.0 - 2.0 * t) def storm_pulse(t, a, b, c, g=1.6): if t < a:
